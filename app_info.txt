# Shiki MX App — информация о приложении и чейнджлог
# Строки, начинающиеся с "#", являются комментариями и игнорируются.

APP_NAME = Shiki MX App
VERSION = 1.4.0
DESCRIPTION = Современное веб-приложение для удобного просмотра профиля, списков аниме и манги, новостей и онлайн-просмотра на базе API Shikimori.
AUTHOR = Shiki MX Team
SHIKIMORI_STATUS = Подключено

# Технологический стек (элементы разделяются точкой с запятой)
STACK = Python 3; Flask; Shikimori REST API; Anicli-API; Vanilla JS; HTML5 / CSS3; Google Sans; SQLite

# Возможности приложения (каждый пункт с новой строки)
[FEATURES]
Интеграция с Shikimori через OAuth 2.0
Авторизация через Google OAuth 2.0
База данных SQLite для хранения пользователей и настроек
Синхронизация настроек пользователя между устройствами (фон, видимость разделов)
Быстрое встроенное кеширование данных и прокси картинок
Поиск аниме и онлайн-просмотр видеоплееров (Kodik, Sibnet)
Полная статистика профиля, список друзей, история и избранное
Адаптивный дизайн с поддержкой тёмной и светлой тем
